option(ROBOT_BUILD_RTOS "Build the Linux FreeRTOS POSIX runtime" ON)
if(ROBOT_BUILD_RTOS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "FreeRTOS POSIX demo requires Linux; use -DROBOT_BUILD_RTOS=OFF for portable tests")
  endif()
  set(KERNEL ${CMAKE_CURRENT_SOURCE_DIR}/vendor/FreeRTOS-Kernel)
  if(NOT EXISTS "${KERNEL}/tasks.c")
    message(FATAL_ERROR "Initialize dependencies: git submodule update --init --recursive")
  endif()
  set(PORT ${KERNEL}/portable/ThirdParty/GCC/Posix)
  find_package(Threads REQUIRED)
  add_library(freertos_kernel STATIC
    ${KERNEL}/tasks.c ${KERNEL}/queue.c ${KERNEL}/list.c
    ${KERNEL}/timers.c ${KERNEL}/event_groups.c ${KERNEL}/stream_buffer.c
    ${KERNEL}/portable/MemMang/heap_4.c
    ${PORT}/port.c ${PORT}/utils/wait_for_event.c)
  target_include_directories(freertos_kernel PUBLIC config ${KERNEL}/include ${PORT})
  target_link_libraries(freertos_kernel PUBLIC Threads::Threads)
  add_executable(robot_demo src/main.cpp)
  target_compile_options(robot_demo PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(robot_demo PRIVATE robot_control freertos_kernel)
  add_executable(mission_demo src/mission_demo.cpp)
  target_compile_options(mission_demo PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(mission_demo PRIVATE robot_mission freertos_kernel)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME rtos_pipeline COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_runtime.py $<TARGET_FILE:robot_demo>)
  set_tests_properties(rtos_pipeline PROPERTIES TIMEOUT 15)
  add_test(NAME rtos_mission COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mission_runtime.py $<TARGET_FILE:mission_demo>)
  set_tests_properties(rtos_mission PROPERTIES TIMEOUT 15)
endif()
